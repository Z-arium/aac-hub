-- The recipe box built from the data stored on the wiki (sync_wiki_data.py), so the crafting API is not
-- asked on every page view. Only the Recipe Cost (it follows the market) comes live: Template:RecipeBox
-- sends it in the argument "cost".
local IconName = require('Module:IconName')
local Store = require('Module:DataStore')

local function trim(s)
	return mw.text.trim(tostring(s or ''))
end

local function esc(s)
	return mw.text.encode(tostring(s or ''), '&lt;&gt;&amp;"\'%[%]{}|')
end

-- Where the box sits on the page (float parameter), inline style
local STYLES = {
	right = 'float:right; clear:right; margin:0 0 0.5em 1em;',
	left = 'float:left; clear:left; margin:0 1em 0.5em 0;',
	center = 'width:fit-content; max-width:100%; margin:0.5em auto;',
}

local function num(n)
	n = tonumber(n)
	if not n then return '' end
	if n == math.floor(n) then return string.format('%d', n) end
	return tostring(n)
end

-- Icon with border, name in the grade colour, amount (what Template:Item shows)
local function itemHtml(frame, name, grade, amount, icon, over)
	grade = trim(grade)
	if grade == '' or grade == 'None' or grade == 'Any' then grade = 'Basic' end
	local out = IconName.icon({ args = { name, grade, file = icon or '', over = over or '' } })
		.. ' ' .. frame:expandTemplate{ title = 'Grade', args = { grade, esc(name) } }
	if amount and amount ~= '' then out = out .. '&amp;nbsp;× ' .. esc(amount) end
	return out
end

local function coins(frame, copper)
	return frame:expandTemplate{ title = 'Coins', args = { num(copper) } }
end

function p.box(frame)
	local a = frame.args
	local r = Store.get('Recipes', a.id)
	if not r or not r.product then return '' end

	local float = trim(a.float):lower()
	local style = STYLES[float] or 'margin:0.5em 0;'
	local root = mw.html.create('div'):addClass('aa-recipebox')
	if float == 'right' or float == 'left' then root:addClass('aa-recipebox-float') end
	root:attr('style', style)

	local tbl = root:tag('table'):addClass('wikitable')
	local function row(label, content)
		local tr = tbl:tag('tr')
		tr:tag('th'):wikitext(label)
		tr:tag('td'):wikitext(content)
	end

	row('Recipe', itemHtml(frame, r.product.name, r.product.grade, num(r.product.amount), r.product.icon, r.product.over))
	if r.labor then row('Labor Cost', '× ' .. num(r.labor)) end
	if r.station then row('Workbench', esc(r.station)) end
	if r.prof then
		local text = esc(r.prof)
		local req = tonumber(r.req)
		if req and req ~= 0 then
			text = text .. '&amp;nbsp;' .. mw.language.getContentLanguage():formatNum(req)
		end
		row('Proficiency', text)
	end
	-- "Chance for 2x": the data says 100 + the extra percent; 10000 and the like are another scale
	local rate = tonumber(r.rate)
	if rate and rate &gt; 100 and rate &lt; 1000 then
		row('Chance for 2x', num(math.floor((rate - 100) * 10 + 0.5) / 10) .. '%')
	end

	local mats = {}
	for _, m in ipairs(r.materials or {}) do
		local grade = trim(m.required)
		if grade == '' or grade == 'Any' then grade = m.grade or '' end
		mats[#mats + 1] = '&lt;div style="margin-bottom:2px;"&gt;' .. itemHtml(frame, m.name, grade, num(m.amount), m.icon, m.over) .. '&lt;/div&gt;'
	end
	local coin = tonumber(r.coin)
	if coin and coin &gt; 0 then
		mats[#mats + 1] = '&lt;div style="margin-top:4px; padding-left:36px; line-height:32px;"&gt;' .. coins(frame, coin) .. '&lt;/div&gt;'
	end
	row('Materials', table.concat(mats))

	-- live from the market; a failed request leaves an error text, then there is no row
	local cost = trim(a.cost)
	if not cost:find('class="error"', 1, true) then
		cost = tonumber(cost)
		if cost and cost &gt; 0 then row('Recipe Cost', coins(frame, cost)) end
	end

	return tostring(root)
end
