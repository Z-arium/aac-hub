-- In-game style card for a skill: icon, skill line, name, element and rank, mana, range, cast time, cooldown,
-- effects granted (buff icons), the description, the damage formula, combos and the chain it belongs to. Data: Module:Data/SkillCards/... (sync_wiki_data.py), or live from the APIs.
-- The hover boxes of mount, glider and pet skills (Module:Mount) are a different thing and stay as they are.
local IconName = require('Module:IconName')
local Store = require('Module:DataStore')
local Consumable = require('Module:Consumable')

local function trim(s)
	return mw.text.trim(tostring(s or ''))
end

local function esc(s)
	return mw.text.encode(tostring(s or ''), '&lt;&gt;&amp;"\'%[%]{}|')
end

local FLOATS = {
	right = 'float:right; clear:right; margin:0 0 0.5em 1em;',
	left = 'float:left; clear:left; margin:0 1em 0.5em 0;',
	center = 'margin:0 auto 0.5em auto;',
}

local function fetched(s)
	s = trim(s)
	if s:find('class="error"', 1, true) then return '' end
	return s
end

local function num(v)
	v = tonumber(v)
	if not v then return 0 end
	return v
end

local function n(x)
	if x == math.floor(x) then return string.format('%d', x) end
	return string.format('%.1f', x)
end

-- 24000 -&gt; "24sec", 90000 -&gt; "1min 30sec", 900000 -&gt; "15min"
local function dur(ms)
	local total = num(ms) / 1000
	local m = math.floor(total / 60)
	local s = total - m * 60
	local out = {}
	if m &gt; 0 then out[#out + 1] = m .. 'min' end
	if s &gt; 0 or m == 0 then out[#out + 1] = n(s) .. 'sec' end
	return table.concat(out, ' ')
end

-- Unfilled placeholders of the game text. #{avg_damage}#{detail_melee_damage} (two in a row) is the damage: the game
-- shows "2180 (0 + 267% Melee Attack)", here the formula "(260% Melee Attack)" in green. Others show "(varies)".
local function holders(s, formulaText)
	s = tostring(s or '')
	local prev
	repeat
		prev = s
		s = s:gsub('(#%b{})%s*%-?%s*#%b{}', '%1')      -- #{a}#{b} and #{min}-#{max}: one value
	until s == prev
	return (s:gsub('#%b{}', function(token)
		if formulaText and token:find('damage', 1, true) then return '|cFF6FD16F(' .. formulaText .. ')|r' end
		return '(varies)'
	end))
end

-- 260 -&gt; "260", 0.5 -&gt; "0.5"
local function g(x)
	x = tonumber(x) or 0
	if x == math.floor(x) then return string.format('%d', x) end
	return (string.format('%.2f', x):gsub('0+$', ''))
end

-- the damage formula as the data gives it: "0 + 260% Melee Attack" or "18-30 + 290% Melee Attack"
local function formula(list)
	local parts = {}
	for _, x in ipairs(type(list) == 'table' and list or {}) do
		local text = g(x.pct) .. '% ' .. trim(x.stat ~= nil and x.stat or 'Attack')
		if x.fmax ~= nil then   -- the flat part, also when it is 0: "(0 + 267% Melee Attack)"
			local flat = g(x.fmin)
			if num(x.fmax) ~= num(x.fmin) then flat = flat .. '-' .. g(x.fmax) end
			text = flat .. ' + ' .. text
		end
		if x.lvl then text = text .. ', scales with level' end
		parts[#parts + 1] = text
	end
	if #parts == 0 then return nil end
	return table.concat(parts, ' + ')
end

-- small icon of the wiki (an effect, a skill)
local function pic(icon, size)
	local f = IconName.iconFile(icon)
	if f == '' then return '' end
	return '[[File:' .. f .. '|' .. size .. 'px|link=]]'
end

-- 1st, 2nd, 3rd, 4th ... 11th, 21st
local function ordinal(i)
	local last, teen = i % 10, i % 100
	local suffix = 'th'
	if teen &lt; 11 or teen &gt; 13 then
		suffix = (last == 1 and 'st') or (last == 2 and 'nd') or (last == 3 and 'rd') or 'th'
	end
	return i .. suffix
end

-- data of the card: stored, or built from the live arguments of Template:Skill
local function data(a)
	local d = Store.get('SkillCards', a.id)
	if d then return d end
	local name = fetched(a.name)
	if name == '' then return nil end
	return { name = name, icon = fetched(a.icon), line = fetched(a.line), desc = fetched(a.desc),
		cd = num(fetched(a.cd)), cast = num(fetched(a.cast)), mana = num(fetched(a.mana)),
		minr = num(fetched(a.minr)), maxr = num(fetched(a.maxr)), rad = num(fetched(a.rad)),
		labor = num(fetched(a.labor)) }
end

-- one line of the box
local function line(root, class, text)
	root:tag('div'):addClass(class):wikitext(text)
end

-- one card. id: the skill ID; setId: prefix of the anchor ids (a chain), links: skills whose card can be opened
local function build(d, id, a, setId, links, reserve)
	local root = mw.html.create('div'):addClass('aa-skill')
	local dmgText = formula(d.dmg)

	-- header: icon, skill line, name (with the rank, like the game); at the right the element (Quake, Life ...)
	local head = root:tag('div'):addClass('aa-skill-head')
	local file = IconName.iconFile(d.icon)
	if file ~= '' then
		head:tag('div'):addClass('aa-skill-icon'):wikitext('[[File:' .. file .. '|48px|link=]]')
	end
	local title = head:tag('div'):addClass('aa-skill-title')
	if trim(d.line) ~= '' then title:tag('div'):addClass('aa-skill-line'):wikitext(esc(d.line)) end
	local name = trim(d.name)
	if num(d.rank) &gt; 0 then name = name .. ' (Rank ' .. n(num(d.rank)) .. ')' end
	title:tag('div'):addClass('aa-skill-name'):wikitext(esc(name))
	if trim(d.elem) ~= '' then
		head:tag('div'):addClass('aa-skill-tag'):tag('div'):addClass('aa-skill-elem'):wikitext(esc(d.elem))
	end

	-- mana, range, effect range, attack percent (only what the data has)
	local mana, maxr, rad = num(d.mana), num(d.maxr), num(d.rad)
	local first = type(d.dmg) == 'table' and d.dmg[1] or nil
	if mana &gt; 0 or maxr &gt; 0 or rad &gt; 0 or first then
		root:tag('div'):addClass('aa-skill-sep')
		if mana &gt; 0 then line(root, 'aa-skill-meta', 'Mana ' .. n(mana)) end
		if maxr &gt; 0 then line(root, 'aa-skill-meta', 'Range: ' .. n(num(d.minr)) .. '-' .. n(maxr) .. ' m') end
		if rad &gt; 0 then line(root, 'aa-skill-meta', 'Effect Range: ' .. n(rad) .. ' m') end
		if first then
			line(root, 'aa-skill-meta', esc(trim(first.stat ~= nil and first.stat or 'Attack') .. ' +' .. g(first.pct) .. '%'))
		end
	end

	-- cast time, cooldown, channeling, labor
	root:tag('div'):addClass('aa-skill-sep')
	local cast = num(d.cast)
	line(root, 'aa-skill-meta', cast &gt; 0 and ('Cast Time ' .. dur(cast)) or 'Instant')
	if num(d.chan) &gt; 0 then line(root, 'aa-skill-meta', 'Channeling ' .. dur(d.chan)) end
	if num(d.cd) &gt; 0 then line(root, 'aa-skill-meta', dur(d.cd) .. ' Cooldown') end
	if num(d.labor) &gt; 0 then line(root, 'aa-skill-meta', 'Labor ' .. n(num(d.labor))) end

	-- effects granted: the icons of the buffs; a box with the name, the effect and the facts (duration, ticks,
	-- stacks) opens when the pointer rests on one. An effect that needs a passive skill says so.
	local shown, needs = {}, {}
	reserve = reserve or {}
	for _, b in ipairs(d.buffs or {}) do
		local f = IconName.iconFile(b.icon)
		if f ~= '' then
			local box = mw.html.create('span'):addClass('aa-skill-buff'):attr('tabindex', '0')
			if b.bad then box:addClass('aa-skill-debuff') end   -- a harmful effect (debuff): red frame
			box:wikitext('[[File:' .. f .. '|32px|link=]]')
			local tip = box:tag('span'):addClass('aa-skill-bufftip')
			tip:tag('span'):addClass('aa-skill-buffname'):wikitext(esc(b.name or ''))
			if type(b.req) == 'table' and trim(b.req.name) ~= '' then
				tip:tag('span'):addClass('aa-skill-reqnote'):wikitext('Requires ' .. esc(b.req.name))
				needs[#needs + 1] = b
			end
			local text = Consumable.paragraphs(holders(b.desc), 'aa-skill-bufftext')
			if text then tip:node(text) end
			local facts = {}
			if num(b.dur) &gt; 0 then facts[#facts + 1] = 'Duration: ' .. dur(b.dur) end
			if num(b.tick) &gt; 0 then facts[#facts + 1] = 'Ticks every ' .. dur(b.tick) end
			if num(b.max) &gt; 1 and not tostring(b.desc or ''):find('Stacks up to', 1, true) then
				facts[#facts + 1] = 'Stacks up to ' .. n(num(b.max)) .. ' times'
			end
			for _, fact in ipairs(facts) do tip:tag('span'):addClass('aa-skill-fact'):wikitext(esc(fact)) end
			shown[#shown + 1] = tostring(box)
		end
	end
	if #shown &gt; 0 or reserve.buffs then
		root:tag('div'):addClass('aa-skill-sep')
		if #shown &gt; 0 then
			line(root, 'aa-skill-meta', 'Effect Granted')
			root:tag('div'):addClass('aa-skill-buffs'):wikitext(table.concat(shown, ' '))
			-- always visible: which passive skill the effect needs (hover for the text of the passive)
			for _, b in ipairs(needs) do
				local row = root:tag('div'):addClass('aa-skill-req'):attr('tabindex', '0')
				row:wikitext(pic(b.req.icon, 20))
				row:tag('span'):addClass('aa-skill-reqtext'):wikitext('Requires ' .. esc(b.req.name))
				local tip = row:tag('span'):addClass('aa-skill-bufftip')
				tip:tag('span'):addClass('aa-skill-buffname'):wikitext(esc(b.req.name))
				local text = Consumable.paragraphs(holders(b.req.desc), 'aa-skill-bufftext')
				if text then tip:node(text) end
			end
		else
			-- another skill of the chain has effects: the same space stays free here, so the cards look alike
			line(root, 'aa-skill-meta aa-skill-reserve', 'Effect Granted')
			root:tag('div'):addClass('aa-skill-reserve aa-skill-buffspace')
		end
	end

	-- description: the damage placeholder becomes the formula, other unfilled ones show "(varies)"
	local body = Consumable.paragraphs(holders(d.desc, dmgText), 'aa-skill-desc')
	if body then
		root:tag('div'):addClass('aa-skill-sep')
		root:node(body)
	end

	-- combos: icon of the condition, arrow and icon of the result, then the sentence of the game
	if type(d.combos) == 'table' and #d.combos &gt; 0 then
		local cbox = root:tag('div'):addClass('aa-skill-combobox')   -- centred in the free space of a stretched card
		cbox:tag('div'):addClass('aa-skill-chainlabel'):tag('span'):wikitext('Combo')
		for _, c in ipairs(d.combos) do
			local row = cbox:tag('div'):addClass('aa-skill-combo')
			local icons = row:tag('span'):addClass('aa-skill-comboicons')
			-- red frame: harmful effect (debuff), green frame: beneficial effect (buff)
			local function framed(icon, kind)
				local span = icons:tag('span')
				if kind == 'bad' then span:addClass('aa-skill-ico-bad') elseif kind == 'good' then span:addClass('aa-skill-ico-good') end
				span:wikitext(pic(icon, 24))
			end
			if trim(c.ci) ~= '' then framed(c.ci, c.ck) end
			if trim(c.ci) ~= '' and trim(c.ri) ~= '' then icons:tag('span'):addClass('aa-skill-arrow'):wikitext('&amp;rarr;') end
			if trim(c.ri) ~= '' then framed(c.ri, c.rk) end
			row:tag('span'):addClass('aa-skill-combotext'):wikitext(esc(holders(c.desc)))
		end
	end

	-- chain: the skills that follow each other; the one shown is outlined, a click on another opens its card
	if type(d.chain) == 'table' and #d.chain &gt; 1 then
		local box = root:tag('div'):addClass('aa-skill-chainbox')   -- sits at the bottom of the card
		box:tag('div'):addClass('aa-skill-chainlabel'):tag('span'):wikitext('Chain')
		local chain = box:tag('div'):addClass('aa-skill-chain')
		-- the icons get smaller the longer the chain is, so that the row stays inside the card (294px inside):
		-- n icons (+ 2px frame) and n - 1 arrows of about 24px. Very long chains wrap onto a second row.
		local count = #d.chain
		local size = math.max(18, math.min(40, math.floor((294 - (count - 1) * 24) / count) - 2))
		for i, c in ipairs(d.chain) do
			if i &gt; 1 then chain:tag('span'):addClass('aa-skill-arrow'):wikitext('&amp;rarr;') end
			local step = chain:tag('span'):addClass('aa-skill-step')
			local icon = '[[File:' .. IconName.iconFile(c.icon) .. '|' .. size .. 'px|link=' ..
				((links and links[tostring(c.id)] and tostring(c.id) ~= tostring(id)) and ('#' .. setId .. c.id) or '') .. ']]'
			if tostring(c.id) == tostring(id) then step:addClass('aa-skill-current') end
			step:tag('span'):addClass('aa-skill-stepicon'):wikitext(IconName.iconFile(c.icon) ~= '' and icon or '')
			step:tag('span'):addClass('aa-skill-steplabel'):wikitext(ordinal(i))
		end
	end
	return root
end

function p.card(frame)
	local a = frame.args
	local d = data(a)
	if not d or trim(d.name) == '' then
		return '&lt;!-- Skill: no data, the API request failed or the skill id is wrong --&gt;'
	end

	local float = trim(a.float):lower()
	if not FLOATS[float] then float = '' end

	-- a chain: every skill of it that is stored gets its own card, only one is shown (CSS :target, see the styles)
	local cards = {}
	local setId, links = nil, nil
	if type(d.chain) == 'table' and #d.chain &gt; 1 then
		links = {}
		for _, c in ipairs(d.chain) do
			local cd = tostring(c.id) == tostring(a.id) and d or Store.get('SkillCards', c.id)
			if cd then links[tostring(c.id)] = cd end
		end
		local count = 0
		for _ in pairs(links) do count = count + 1 end
		if count &gt; 1 then
			setId = 'aa-skill' .. trim(a.id) .. '-'   -- different for every card on a page (a counter did not survive between the cards)
		else
			links = nil
		end
	end

	-- does a skill of the chain have effects? Then every card keeps that space (a "Requires ..." line is not kept:
	-- the card that has it is just a bit taller)
	local reserve = nil
	if links then
		reserve = {}
		for _, cd in pairs(links) do
			for _, b in ipairs(cd.buffs or {}) do
				if IconName.iconFile(b.icon) ~= '' then reserve.buffs = true end
			end
		end
	end

	local wrap = mw.html.create('div'):addClass('aa-skill-set')
	if float == 'right' or float == 'left' then wrap:addClass('aa-skill-float') end
	if float == 'right' then wrap:addClass('aa-skill-right') end
	if float ~= '' then wrap:attr('style', FLOATS[float]) end
	if setId then
		-- the other cards first, the requested one last: ":target ~ .aa-skill-main" hides it when another is chosen
		for _, c in ipairs(d.chain) do
			if links[tostring(c.id)] and tostring(c.id) ~= tostring(a.id) then
				wrap:tag('span'):addClass('aa-skill-anchor'):attr('id', setId .. c.id)
				wrap:node(build(links[tostring(c.id)], c.id, a, setId, links, reserve))
			end
		end
		wrap:tag('span'):addClass('aa-skill-anchor'):attr('id', setId .. a.id)
		wrap:node(build(d, a.id, a, setId, links, reserve):addClass('aa-skill-main'))
	else
		wrap:node(build(d, a.id, a, nil, nil):addClass('aa-skill-main'))
	end
	return tostring(wrap)
end
