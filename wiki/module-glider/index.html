Module:Glider

Source code (Scribunto) as stored on the wiki.

local p = {}

local IconName = require('Module:IconName')
local Store = require('Module:DataStore')   -- data stored on the wiki by sync_wiki_data.py
local Mount = require('Module:Mount')       -- its hover box for a skill
-- Optional manual list: skill name -> skill ID. Only used for skills the API did not give an ID for.
local okIds, SkillIds = pcall(require, 'Module:GliderSkillIds')
if not okIds or type(SkillIds) ~= 'table' then SkillIds = {} end

-- "Glider Charge" and "glider charge!" are the same skill
local function key(s)
	return (tostring(s or ''):lower():gsub('[^%a%d]', ''))
end

-- The skills as sent by Template:Glider: "id¦name‖id¦name‖..." -> key(name) -> id
local function parseSkillIds(str)
	str = tostring(str or '')
	local ids = {}
	if str:find('class="error"', 1, true) then return ids end
	for _, row in ipairs(mw.text.split(str, '‖', true)) do
		local f = mw.text.split(row, '¦', true)
		local id, name = tonumber(mw.text.trim(f[1] or '')), mw.text.trim(f[2] or '')
		if id and name ~= '' and not ids[key(name)] then ids[key(name)] = math.floor(id) end
	end
	return ids
end

-- Make text safe to put inside HTML / wikitext
local function esc(s)
	return mw.text.encode(tostring(s or ''), '<>&"\'%[%]{}|')
end

local function trim(s)
	return mw.text.trim(tostring(s or ''))
end

-- Colours of the game's short codes: |nc;...|r is orange, |ni;...|r is yellow ("Immune to |ni;Turbulence|r").
-- Other short codes (|nd; |nr; ...) are shown without colour.
local NCODE = { c = 'FF9C27', i = 'F5CB65', d = 'A3E3FD' }

-- The game marks highlighted words as |cFFRRGGBB...|r.
-- Only those parts get a colour, the rest keeps the card's text colour.
local function coloured(text)
	text = tostring(text or '')
	text = text:gsub('|n(%a);(.-)|r', function(code, inner)
		if NCODE[code] then return '|cFF' .. NCODE[code] .. inner .. '|r' end
		return inner
	end)
	local out, pos = {}, 1
	while true do
		local s, e, rgb, inner = text:find('|c%x%x(%x%x%x%x%x%x)(.-)|r', pos)
		if not s then break end
		out[#out + 1] = esc(text:sub(pos, s - 1))
		out[#out + 1] = '<span style="color:#' .. rgb .. '">' .. esc(inner) .. '</span>'
		pos = e + 1
	end
	out[#out + 1] = esc(text:sub(pos))
	return table.concat(out)
end

-- Where the card sits on the page (float parameter), set inline so it works without the style page
local FLOATS = {
	right = 'float:right; clear:right; margin:0 0 0.5em 1em;',
	left = 'float:left; clear:left; margin:0 1em 0.5em 0;',
	center = 'margin:0 auto 0.5em auto;',
}

-- The "Use:" texts of the game (not part of the API). Magithopters have their own text.
local USE_GLIDER = 'Equip a glider to glide from place to place. You can only use glider skills while airborne and will be knocked off your glider if attacked.'
local USE_MAGITHOPTER = 'Equip wings to use Free Glide. You can only use glider skills while airborne, and will be knocked off your glider if you are struck by an effect that prevents movement. You will remain airborne if attacked. Gliding disables Parry, Evasion, and Shields; increases Received damage +50%.'

-- key(name) -> id for the skills of a stored glider
local function storedSkillIds(stored)
	local ids = {}
	for _, s in ipairs(stored.skills or {}) do
		if s.id and s.name and not ids[key(s.name)] then ids[key(s.name)] = math.floor(s.id) end
	end
	return ids
end

-- The stat lines of a glider description. Some descriptions (Arkana Wings) have no blank line between the
-- story and the stats, or even write the stats after each other on one line, so a line that starts with one
-- of these labels is where the stats begin.
local LABELS = { 'Flight Speed', 'Air Speed', 'Gliding Ability', 'Cruising Ability', 'Turning Speed',
	'Sheering Speed', 'Launch Height', 'Ascension', 'Special Skills?', 'Special Note' }

local function startsWithLabel(line)
	for _, l in ipairs(LABELS) do
		if line:find('^' .. l .. ':') then return true end
	end
	return false
end

-- puts every label on its own line
local function splitStats(desc)
	for _, l in ipairs(LABELS) do
		desc = desc:gsub('(%S)[ \t]+(' .. l .. ':)', '%1\n%2')
	end
	return desc
end

-- "Special Skill: Glider Charge, Instant Acceleration" -> one row per skill.
-- A skill with an ID (from the API, or Module:GliderSkillIds) gets a hover box.
local function skillRow(frame, label, value, apiIds)
	local row = mw.html.create('div'):addClass('aa-glider-skillrow')
	row:tag('span'):addClass('aa-glider-skilllabel'):wikitext(esc(label) .. '&#32;')
	local box = row:tag('div'):addClass('aa-glider-skills')

	local function add(rgb, name)
		name = trim(name)
		if name == '' then return end
		local html
		local id = apiIds[key(name)] or SkillIds[name]
		local sk = id and Store.get('Skills', id)
		if sk then
			-- the skill's data is stored on the wiki: no request needed
			html = Mount.hover({ args = { name = name, realname = sk.name or '', color = rgb or '', cd = sk.cd or '',
				cast = sk.cast or '', mana = sk.mana and tostring(sk.mana) or '', desc = sk.desc or '', icon = sk.icon or '' } })
		elseif id then
			html = frame:expandTemplate{ title = 'GliderSkill', args = { id = tostring(id), name = name, color = rgb or '' } }
		elseif rgb then
			html = '<span style="color:#' .. rgb .. '">' .. esc(name) .. '</span>'
		else
			html = esc(name)
		end
		box:tag('div'):wikitext(html)
	end

	local found = false
	for rgb, inner in value:gmatch('|c%x%x(%x%x%x%x%x%x)(.-)|r') do
		found = true
		-- the data may colour several skills as one piece ("Rocket Launch, Divebomb")
		for _, name in ipairs(mw.text.split(inner, ',', true)) do add(rgb, name) end
	end
	if not found then   -- no colour codes in the data: split at the commas
		for _, name in ipairs(mw.text.split(value, ',', true)) do add(nil, name) end
	end
	return tostring(row)
end

-- A failed API request (with "suppress error" in the template) leaves External Data's own error text
-- behind. Treat that as "no data" instead of showing it.
local function fetched(s)
	s = trim(s)
	if s:find('class="error"', 1, true) then return '' end
	return s
end

function p.card(frame)
	local a = frame.args
	local stored = Store.get('Gliders', a.id)
	-- live data: read the skill list first, the hover boxes fetch data of their own later
	local apiIds = stored and storedSkillIds(stored) or parseSkillIds(a.skills)
	local name = stored and trim(stored.name) or fetched(a.name)
	if name == '' then
		return '<!-- Glider: no data, the API request failed or the item id is wrong -->'
	end
	local grade = stored and trim(stored.grade) or fetched(a.grade)
	local hasGrade = not (grade == '' or grade == 'None' or grade == 'Any')
	if not hasGrade then grade = 'Basic' end   -- no grade: Basic look, and no grade line
	local desc = (stored and trim(stored.desc) or fetched(a.desc)):gsub('\r', '')
	local float = trim(a.float):lower()
	if not FLOATS[float] then float = '' end
	local isMagi = desc:find('Air Speed', 1, true) ~= nil
	local category = stored and trim(stored.category) or fetched(a.category)
	if isMagi then category = 'Magithopter' elseif category == '' then category = 'Glider' end
	local use = trim(a.use)
	if use == '' then use = isMagi and USE_MAGITHOPTER or USE_GLIDER end

	local root = mw.html.create('div'):addClass('aa-glider')
	if float == 'right' or float == 'left' then root:addClass('aa-glider-float') end
	if float == 'right' then root:addClass('aa-glider-right') end
	if float ~= '' then root:attr('style', FLOATS[float]) end

	-- header: icon, category, grade, name
	local head = root:tag('div'):addClass('aa-glider-head')
	head:tag('div'):addClass('aa-glider-icon'):wikitext(IconName.icon({ args = { name, grade, size = 48, file = stored and trim(stored.icon) or fetched(a.icon) } }))
	local title = head:tag('div'):addClass('aa-glider-title')
	title:tag('div'):addClass('aa-glider-cat'):wikitext(esc(category))
	if hasGrade then
		title:tag('div'):addClass('aa-glider-grade'):wikitext(
			frame:expandTemplate{ title = 'Grade', args = { grade, esc(grade) } })
	end
	title:tag('div'):addClass('aa-glider-name'):wikitext(
		frame:expandTemplate{ title = 'Grade', args = { grade, esc(name) } })

	root:tag('div'):addClass('aa-glider-sep')

	-- description: the first paragraph is running text (the data may break it into short lines),
	-- every line after the blank line (Flight Speed, Special Skill ...) stays on its own line
	local lines = mw.text.split(splitStats(desc), '\n', true)
	local intro, i = {}, 1
	while i <= #lines and trim(lines[i]) ~= '' and not startsWithLabel(trim(lines[i])) do
		intro[#intro + 1] = trim(lines[i])
		i = i + 1
	end
	if #intro > 0 then
		root:tag('div'):addClass('aa-glider-desc'):wikitext(coloured(table.concat(intro, ' ')))
	end
	local stats = root:tag('div'):addClass('aa-glider-stats')
	for j = i, #lines do
		local l = trim(lines[j])
		if l ~= '' then
			local label, value = l:match('^(Special Skills?:)%s*(.*)$')
			if label then
				stats:wikitext(skillRow(frame, label, value, apiIds))
			else
				-- "Special Note: Press S or ... longer. Fly in stealth for 3 seconds when Ascending."
				-- The sentence after the standard note is an effect of this glider: it gets its own line.
				local note, rest = l:match('^(Special Note:%s*Press S[^%.]*%.)%s+(%S.*)$')
				if note then
					stats:tag('div'):wikitext(coloured(note))
					stats:tag('div'):wikitext(coloured(rest))
				else
					stats:tag('div'):wikitext(coloured(l))
				end
			end
		end
	end

	root:tag('div'):addClass('aa-glider-label'):wikitext('Use:')
	root:tag('div'):addClass('aa-glider-green'):wikitext(esc(use))
	root:tag('div'):addClass('aa-glider-label'):wikitext('Equip effect')
	root:tag('div'):addClass('aa-glider-green'):wikitext("You can't carry another trade pack.")

	return tostring(root)
end

return p
