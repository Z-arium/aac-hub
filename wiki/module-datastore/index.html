Module:DataStore

Source code (Scribunto) as stored on the wiki.

local p = {}

-- Reads the data that sync_wiki_data.py stored on the wiki (Module:Data/<Kind>/<number>) so the templates
-- do not have to ask the APIs again and again.
local IconName = require('Module:IconName')

-- How many IDs share one data page. KEEP THIS IN SYNC with SHARD in sync_wiki_data.py
local SHARD = { Mounts = 100, Gliders = 100, Skills = 1000, Items = 1000, Recipes = 100, Cards = 100, Pets = 100, SkillCards = 100 }

local function trim(s)
	return mw.text.trim(tostring(s or ''))
end

local function esc(s)
	return mw.text.encode(tostring(s or ''), '<>&"\'%[%]{}|')
end

local loaded = {}

-- One stored entry, or nil when it is not stored (yet)
function p.get(kind, id)
	id = tonumber(id)
	local size = SHARD[kind]
	if not id or not size then return nil end
	id = math.floor(id)
	local key = kind .. '/' .. math.floor(id / size)
	if loaded[key] == nil then
		local ok, data = pcall(require, 'Module:Data/' .. key)
		loaded[key] = (ok and type(data) == 'table') and data or false
	end
	local data = loaded[key]
	return data and data[id] or nil
end

-- {{#invoke:DataStore|has|Mounts|8001101}} -> "1" when the data is stored, otherwise nothing
function p.has(frame)
	return p.get(trim(frame.args[1]), frame.args[2]) and '1' or ''
end

-- An item by ID, for {{Item|ItemID=...}}: icon with grade border, name in the grade colour, amount.
-- Args: id, amount (optional), grade (optional, replaces the item's own grade)
function p.item(frame)
	local a = frame.args
	local d = p.get('Items', a.id)
	if not d or trim(d.name) == '' then return '' end
	local grade = trim(a.grade)
	if grade == '' then grade = trim(d.grade) end
	if grade == '' or grade == 'None' or grade == 'Any' then grade = 'Basic' end
	local out = IconName.icon({ args = { d.name, grade, file = d.icon or '', over = d.over or '' } })
		.. ' ' .. frame:expandTemplate{ title = 'Grade', args = { grade, esc(d.name) } }
	local amount = trim(a.amount)
	if amount ~= '' then out = out .. '&nbsp;× ' .. esc(amount) end
	return out
end

return p
