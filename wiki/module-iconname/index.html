Module:IconName

Source code (Scribunto) as stored on the wiki.

local p = {}

-- Grade name -> number of the border image (File:Grade1.png ... File:Grade11.png)
local GRADES = {
	Basic = 1, Grand = 2, Rare = 3, Arcane = 4, Heroic = 5, Unique = 6,
	Celestial = 7, Divine = 8, Epic = 9, Legendary = 10, Mythic = 11,
}

-- Characters the wiki does not accept in file names, and what is used instead.
-- KEEP THIS IN SYNC with the REPLACE table in upload_icons.py
local CHARS = {
	[':'] = '-', ['['] = '(', [']'] = ')', ['#'] = '-',
	['<'] = '-', ['>'] = '-', ['|'] = '-', ['{'] = '(', ['}'] = ')',
	['/'] = '-', ['\\'] = '-',   -- game icons in a folder: "aac/taming_bag_t3.png" -> File:Aac-taming_bag_t3.png
}

local function fileName(s)
	return (mw.ustring.gsub(s, '.', function(c) return CHARS[c] end))
end

-- Icon file name from the game data ("icon_item_3400.png", "icon_item_3400.dds", "aac/taming_bag_t3.png")
-- -> the name of the file on the wiki. Empty when there is no file name.
function p.iconFile(name)
	name = mw.text.trim(tostring(name or ''))
	if name == '' or name == 'null' then return '' end
	name = name:gsub('%.%w+$', '') .. '.png'
	return fileName(name)
end

-- Item name -> icon file name part
function p.name(frame)
	return fileName(mw.text.trim(frame.args[1] or ''))
end

-- Icon with the grade border painted on top.
-- {{#invoke:IconName|icon|Item name|Grade|size=32}}
-- No grade, "Any", "None", "Crude" or unknown -> Basic border (Grade1.png).
-- file=icon_item_3400.png (optional): the game's own icon file name for the item. It is used when given,
-- otherwise the file named after the item (the older way).
-- over=top_seal_06.png (optional): the game's overlay picture for the item ("over_icon_filename"), painted on the icon
-- under the grade border (sealed items, unconfirmed items ...).
-- No "does the file exist" check on purpose: such checks are "expensive function calls" and a page is only
-- allowed about 100 of them, which a page with several cards soon passes.
function p.icon(frame)
	local args = frame.args
	local name = mw.text.trim(args[1] or '')
	local grade = mw.text.trim(args[2] or '')
	local size = tonumber(args.size) or 32
	local file = fileName(name)
	local own = p.iconFile(args.file)
	if own ~= '' then
		file = own:gsub('%.png$', '')
	end

	-- "epic" / "EPIC" -> "Epic"
	grade = grade:sub(1, 1):upper() .. grade:sub(2):lower()

	local box = mw.html.create('span')
		:css('position', 'relative')
		:css('display', 'inline-block')
		:css('width', size .. 'px')
		:css('height', size .. 'px')
		:css('line-height', '0')
		:css('vertical-align', 'middle')
		:wikitext('[[File:' .. file .. '.png|' .. size .. 'px|link=]]')

	local over = p.iconFile(args.over)
	if over ~= '' then
		box:tag('span')
			:css('position', 'absolute')
			:css('top', '0')
			:css('left', '0')
			:wikitext('[[File:' .. over .. '|' .. size .. 'px|link=]]')
	end

	-- No grade (empty, "None", "Any", "Crude" ...) gets the Basic border, like in the game
	local n = GRADES[grade] or 1
	if n then
		box:tag('span')
			:css('position', 'absolute')
			:css('top', '0')
			:css('left', '0')
			:wikitext('[[File:Grade' .. n .. '.png|' .. size .. 'px|link=]]')
	end

	return tostring(box)
end

return p
