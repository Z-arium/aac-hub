local ROW, COL = '‖', '¦'   -- separators used by Template:Mount to hand the skill list over

-- Make text safe to put inside HTML / wikitext
local function esc(s)
	return mw.text.encode(tostring(s or ''), '&lt;&gt;&amp;"\'%[%]{}|')
end

local function trim(s)
	return mw.text.trim(tostring(s or ''))
end

-- "10.0" -&gt; "10"
local function int(s)
	local n = tonumber(s)
	if n then return tostring(n) end
	s = trim(s)
	if s == 'null' then return '' end
	return s
end

-- Remove colour codes we do not use
local function clean(s)
	return (s:gsub('|c%x%x%x%x%x%x%x%x', ''):gsub('|r', ''))
end

-- Text outside colour codes. The game also highlights durations it fills in itself, which the data
-- writes as "10s" (written-out times such as "3 seconds" are not highlighted). Shown as "10sec".
local function plain(s)
	s = clean(s)
	local out, pos = {}, 1
	while true do
		local a, b, num, unit = s:find('(%d[%d%.]*)([sm])%f[%A]', pos)
		if not a then break end
		local before = a &gt; 1 and s:sub(a - 1, a - 1) or ''
		local isTime = (unit == 's') or (unit == 'm' and s:sub(math.max(1, a - 4), a - 1) == 'for ')
		if before:find('%w') or not isTime then
			out[#out + 1] = esc(s:sub(pos, b))      -- inside a word, or a distance like 25m: leave it
		else
			out[#out + 1] = esc(s:sub(pos, a - 1))
			out[#out + 1] = '&lt;span style="color:#FF9C27"&gt;' .. esc(num) .. (unit == 's' and 'sec' or 'min') .. '&lt;/span&gt;'
		end
		pos = b + 1
	end
	out[#out + 1] = esc(s:sub(pos))
	return table.concat(out)
end

-- Effect text. The game marks the values it highlights as |cFFRRGGBB...|r, e.g. "|cFFFF9C27+80%|r".
-- Only those parts get a colour. Unfilled placeholders such as #{avg_damage} become "(varies)".
local NCODE = { c = 'FF9C27', i = 'F5CB65', d = 'A3E3FD' }   -- |nc;| orange, |ni;| yellow, |nd;| light blue; other |n?; codes are dropped

local function effectHtml(text)
	text = tostring(text or ''):gsub('#%b{}', '(varies)')
	text = text:gsub('|n(%a);(.-)|r', function(code, inner)
		if NCODE[code] then return '|cFF' .. NCODE[code] .. inner .. '|r' end
		return inner
	end)
	local out, pos = {}, 1
	while true do
		local s, e, rgb, inner = text:find('|c%x%x(%x%x%x%x%x%x)(.-)|r', pos)
		if not s then break end
		out[#out + 1] = plain(text:sub(pos, s - 1))
		out[#out + 1] = '&lt;span style="color:#' .. rgb .. '"&gt;' .. esc(clean(inner)) .. '&lt;/span&gt;'
		pos = e + 1
	end
	out[#out + 1] = plain(text:sub(pos))
	return (table.concat(out):gsub('\n', '&lt;br&gt;'))
end

-- The skills as sent by Template:Mount:
-- "level¦name¦effect¦cooldown¦casting¦mana¦effect with colour codes¦icon file‖level¦name¦..."
-- Result: ordered list of { level = "30", names = {...} } and a table name -&gt; { effect, cooldown, casting, mana }
local function parseSkillList(str)
	local groups, byLevel, info = {}, {}, {}
	for _, row in ipairs(mw.text.split(tostring(str or ''), ROW, true)) do
		local f = mw.text.split(row, COL, true)
		local name = trim(f[2])
		if name ~= '' then
			local lv = int(f[1])
			local g = (lv ~= '') and byLevel[lv] or nil   -- skills without a level each get their own line
			if not g then
				g = { level = lv, names = {} }
				if lv ~= '' then byLevel[lv] = g end
				groups[#groups + 1] = g
			end
			g.names[#g.names + 1] = name
			local coloured = trim(f[7])
			info[name] = { effect = coloured ~= '' and coloured or trim(f[3]),
				cooldown = trim(f[4]), casting = trim(f[5]), mana = trim(f[6]), icon = trim(f[8]) }
		end
	end
	return groups, info
end

-- "30s" -&gt; "30sec"
local function secs(s)
	return (s:gsub('(%d)s%f[%A]', '%1sec'):gsub('(%d)m%f[%A]', '%1min'))
end

-- Fallback: cut skills out of the plain description ("Lv30: Overwhelm or Drop Back").
-- Returns the intro lines and the skill groups (no effect texts).
local function parseDescription(text)
	text = (tostring(text or ''):gsub('\r', ''))
	local starts = {}
	-- "Lv20: Name" and "Level 20: Name" (some mounts, e.g. the candy-fueled snails, write it out)
	for _, word in ipairs({ 'Lv', 'Level' }) do
		local init = 1
		while true do
			local s, e, lv, colon = text:find(word .. '%s*(%d+)%s*(:?)', init)
			if not s then break end
			init = e + 1
			-- the data sometimes forgets the colon ("Lv20 Name"), accepted at the start of a line
			local lineStart = (s == 1) or text:sub(s - 1, s - 1) == '\n'
			if colon == ':' or lineStart then
				starts[#starts + 1] = { s = s, e = e, lv = lv }
			end
		end
	end
	table.sort(starts, function(a, b) return a.s &lt; b.s end)

	local intro, groups = text, {}
	if #starts &gt; 0 then
		intro = text:sub(1, starts[1].s - 1)
		for i, st in ipairs(starts) do
			local stop = starts[i + 1] and (starts[i + 1].s - 1) or #text
			local spec = trim(text:sub(st.e + 1, stop))
			local names = {}
			for _, n in ipairs(mw.text.split(spec, ' or ', true)) do
				n = trim(n)
				if n ~= '' then names[#names + 1] = n end
			end
			groups[#groups + 1] = { level = st.lv, names = names }
		end
	end

	local lines = {}
	for _, l in ipairs(mw.text.split(intro, '\n', true)) do
		l = trim(l)
		if l ~= '' then lines[#lines + 1] = l end
	end
	return lines, groups
end

-- A failed API request (with "suppress error" in the template) leaves External Data's own error text
-- behind. Treat that as "no data" instead of showing it.
local function fetched(s)
	s = trim(s)
	if s:find('class="error"', 1, true) then return '' end
	return s
end

-- The skills of a stored mount in the same text form Template:Mount hands over for live data:
-- "level¦name¦effect¦cooldown¦casting¦mana¦effect with colour codes¦icon‖..."
local function storedSkillString(stored)
	local rows = {}
	for _, s in ipairs(stored.skills or {}) do
		local sk = Store.get('Skills', s.id) or {}
		rows[#rows + 1] = table.concat({ s.lv and tostring(s.lv) or '', s.name or '', s.desc or '',
			sk.cd or '', sk.cast or '', sk.mana and tostring(sk.mana) or '', sk.desc or '', sk.icon or '' }, COL) .. ROW
	end
	return table.concat(rows)
end

			descLevel[k] = dg.level
			descOrder[k] = i
			descGroup[k] = gi
			descKeys[#descKeys + 1] = k
		end
	end
	-- exact match on the simplified name, otherwise the closest name that is at most 2 letters off
	-- (the description misspells some names, e.g. "Release Poision"). Returns the description's key.
	local function findKey(name)
		local k = key(name)
		if descLevel[k] then return k end
		local best, bestD = nil, 3
		for _, dk in ipairs(descKeys) do
			local d = distance(k, dk)
			if d &lt; bestD then best, bestD = dk, d end
		end
		return best
	end

	-- Several skills of one level are "A or B" (a choice) only when the description writes them on one
	-- line. When it lists the same level on several lines ("Lv20: A" / "Lv20: B", like Carrot Dash),
	-- the mount gets all of them, so every description line becomes its own line here too.
	local split = {}
	for _, g in ipairs(groups) do
		if g.level ~= '' and (descCount[g.level] or 0) &gt; 1 then
			local parts, order = {}, {}
			for _, n in ipairs(g.names) do
				local dk = findKey(n)
				local gi = dk and descGroup[dk] or descFirst[g.level]
				if not parts[gi] then parts[gi] = {}; order[#order + 1] = gi end
				parts[gi][#parts[gi] + 1] = n
			end
			table.sort(order)
			for _, gi in ipairs(order) do
				split[#split + 1] = { level = g.level, names = parts[gi], dg = gi }
			end
		else
			g.dg = descFirst[g.level]
			split[#split + 1] = g
		end
	end
	groups = split

	local merged, byDG = {}, {}
	for _, g in ipairs(groups) do
		if g.level ~= '' then
			if g.dg then byDG[g.dg] = g end
			merged[#merged + 1] = g
		end
	end
	local placed = false
	for _, g in ipairs(groups) do
		if g.level == '' then
			for _, n in ipairs(g.names) do
				local dk = findKey(n)
				local gi = dk and descGroup[dk]
				local target = gi and byDG[gi]
				if target then
					target.names[#target.names + 1] = n
					descOrder[key(n)] = descOrder[dk]
					placed = true
				elseif gi then
					-- the description has a line of its own for this skill (e.g. "Lv20 Powerful Wingbeats")
					local ng = { level = fallbackGroups[gi].level, names = { n }, dg = gi }
					byDG[gi] = ng
					merged[#merged + 1] = ng
					placed = true
				elseif #fallbackGroups &gt; 0 and introKeys[key(n)] then
					-- listed in the description without a level: keep it, as the first row
					merged[#merged + 1] = { level = '', names = { n }, keep = true }
					placed = true
				else
					merged[#merged + 1] = { level = '', names = { n } }
				end
			end
		end
	end
	if placed then
		for _, g in ipairs(merged) do
			table.sort(g.names, function(a, b)
				return (descOrder[key(a)] or 99) &lt; (descOrder[key(b)] or 99)
			end)
		end
		-- keep the rows in level order (same level: description order), level-less ones last
		for i, g in ipairs(merged) do g.idx = i end
		table.sort(merged, function(a, b)
			local la, lb = lvl(a), lvl(b)
			if la ~= lb then return la &lt; lb end
			if (a.dg or 99) ~= (b.dg or 99) then return (a.dg or 99) &lt; (b.dg or 99) end
			return a.idx &lt; b.idx
		end)
		groups = merged
	end

	-- The description lists the mount's skills by level (like the game's tooltip). Then show them in that
	-- order (the API's order is arbitrary) and leave out skills that are on no level and not in the
	-- description (basic skills such as "Mounted Attack", which the game does not list either).
	if #fallbackGroups &gt; 0 then
		local kept = {}
		for _, g in ipairs(groups) do
			if g.level ~= '' or g.keep then kept[#kept + 1] = g end
		end
		if #kept &gt; 0 then
			for i, g in ipairs(kept) do g.idx = i end
			table.sort(kept, function(a, b)
				local la, lb = lvl(a), lvl(b)
				if la ~= lb then return la &lt; lb end
				if (a.dg or 99) ~= (b.dg or 99) then return (a.dg or 99) &lt; (b.dg or 99) end
				return a.idx &lt; b.idx
			end)
			groups = kept
		end
	end

	local isSkill = {}
	for _, g in ipairs(groups) do
		for _, n in ipairs(g.names) do isSkill[key(n)] = true end
	end
	-- The API breaks long descriptions into short lines in the middle of sentences. Glue a line onto
	-- the previous one unless that one ended a sentence (. ! ? : or a closing quote).
	local paras = {}
	for _, l in ipairs(lines) do
		if not isSkill[key(l)] then   -- the description may list the skill names again, we show them below
			local last = paras[#paras]
			if last and not last:find('[%.!%?:"\'%)]$') then
				paras[#paras] = last .. ' ' .. l
			else
				paras[#paras + 1] = l
			end
		end
	end
	for _, para in ipairs(paras) do
		root:tag('div'):addClass('aa-mount-desc'):wikitext(esc(para))
	end
	for _, g in ipairs(groups) do
		local line = root:tag('div'):addClass('aa-mount-skill')
		if g.level ~= '' then
			line:tag('span'):addClass('aa-mount-lv'):wikitext('Lv' .. esc(g.level))
			line:wikitext(': ')
		end
		for i, n in ipairs(g.names) do
			if i &gt; 1 then line:tag('span'):addClass('aa-mount-or'):wikitext(' or ') end
			line:node(skillNode(n, skills))
		end
	end

	-- use, speed
	root:tag('div'):addClass('aa-mount-use'):wikitext('Use:')
	root:tag('div'):addClass('aa-mount-usetext'):wikitext('Summons a mount.')
	if speed ~= '' then
		local line = root:tag('div'):addClass('aa-mount-speed')
		line:wikitext('Base Move Speed ')
		line:tag('span'):addClass('aa-mount-speedval'):wikitext(esc(speed) .. 'm/s')
	end

	return tostring(root)
end
