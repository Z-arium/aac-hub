Source code (Scribunto) as stored on the wiki.

local p = {}

local IconName = require('Module:IconName')
local Store = require('Module:DataStore')   -- data stored on the wiki by sync_wiki_data.py

-- Skill icons straight from aa-classic.com (the exact game icon of each skill).
-- MediaWiki only shows images from other sites when the wiki owner allows it
-- ($wgAllowExternalImagesFrom). Set this to true once that works, see Template:Mount docs.
local EXTERNAL_ICONS = false
local ICON_BASE = 'https://aa-classic.com/game/icons/'

local ROW, COL = '‖', '¦'   -- separators used by Template:Mount to hand the skill list over

-- Make text safe to put inside HTML / wikitext
local function esc(s)
	return mw.text.encode(tostring(s or ''), '<>&"\'%[%]{}|')
end

local function trim(s)
	return mw.text.trim(tostring(s or ''))
end

-- "10.0" -> "10"
local function int(s)
	local n = tonumber(s)
	if n then return tostring(n) end
	s = trim(s)
	if s == 'null' then return '' end
	return s
end

-- Remove colour codes we do not use
local function clean(s)
	return (s:gsub('|c%x%x%x%x%x%x%x%x', ''):gsub('|r', ''))
end

-- Text outside colour codes. The game also highlights durations it fills in itself, which the data
-- writes as "10s" (written-out times such as "3 seconds" are not highlighted). Shown as "10sec".
local function plain(s)
	s = clean(s)
	local out, pos = {}, 1
	while true do
		local a, b, num, unit = s:find('(%d[%d%.]*)([sm])%f[%A]', pos)
		if not a then break end
		local before = a > 1 and s:sub(a - 1, a - 1) or ''
		local isTime = (unit == 's') or (unit == 'm' and s:sub(math.max(1, a - 4), a - 1) == 'for ')
		if before:find('%w') or not isTime then
			out[#out + 1] = esc(s:sub(pos, b))      -- inside a word, or a distance like 25m: leave it
		else
			out[#out + 1] = esc(s:sub(pos, a - 1))
			out[#out + 1] = '<span style="color:#FF9C27">' .. esc(num) .. (unit == 's' and 'sec' or 'min') .. '</span>'
		end
		pos = b + 1
	end
	out[#out + 1] = esc(s:sub(pos))
	return table.concat(out)
end

-- Effect text. The game marks the values it highlights as |cFFRRGGBB...|r, e.g. "|cFFFF9C27+80%|r".
-- Only those parts get a colour. Unfilled placeholders such as #{avg_damage} become "(varies)".
local NCODE = { c = 'FF9C27', i = 'F5CB65', d = 'A3E3FD' }   -- |nc;| orange, |ni;| yellow, |nd;| light blue; other |n?; codes are dropped

local function effectHtml(text)
	text = tostring(text or ''):gsub('#%b{}', '(varies)')
	text = text:gsub('|n(%a);(.-)|r', function(code, inner)
		if NCODE[code] then return '|cFF' .. NCODE[code] .. inner .. '|r' end
		return inner
	end)
	local out, pos = {}, 1
	while true do
		local s, e, rgb, inner = text:find('|c%x%x(%x%x%x%x%x%x)(.-)|r', pos)
		if not s then break end
		out[#out + 1] = plain(text:sub(pos, s - 1))
		out[#out + 1] = '<span style="color:#' .. rgb .. '">' .. esc(clean(inner)) .. '</span>'
		pos = e + 1
	end
	out[#out + 1] = plain(text:sub(pos))
	return (table.concat(out):gsub('\n', '<br>'))
end

-- The skills as sent by Template:Mount:
-- "level¦name¦effect¦cooldown¦casting¦mana¦effect with colour codes¦icon file‖level¦name¦..."
-- Result: ordered list of { level = "30", names = {...} } and a table name -> { effect, cooldown, casting, mana }
local function parseSkillList(str)
	local groups, byLevel, info = {}, {}, {}
	for _, row in ipairs(mw.text.split(tostring(str or ''), ROW, true)) do
		local f = mw.text.split(row, COL, true)
		local name = trim(f[2])
		if name ~= '' then
			local lv = int(f[1])
			local g = (lv ~= '') and byLevel[lv] or nil   -- skills without a level each get their own line
			if not g then
				g = { level = lv, names = {} }
				if lv ~= '' then byLevel[lv] = g end
				groups[#groups + 1] = g
			end
			g.names[#g.names + 1] = name
			local coloured = trim(f[7])
			info[name] = { effect = coloured ~= '' and coloured or trim(f[3]),
				cooldown = trim(f[4]), casting = trim(f[5]), mana = trim(f[6]), icon = trim(f[8]) }
		end
	end
	return groups, info
end

-- "30s" -> "30sec"
local function secs(s)
	return (s:gsub('(%d)s%f[%A]', '%1sec'):gsub('(%d)m%f[%A]', '%1min'))
end

-- Fallback: cut skills out of the plain description ("Lv30: Overwhelm or Drop Back").
-- Returns the intro lines and the skill groups (no effect texts).
local function parseDescription(text)
	text = (tostring(text or ''):gsub('\r', ''))
	local starts = {}
	-- "Lv20: Name" and "Level 20: Name" (some mounts, e.g. the candy-fueled snails, write it out)
	for _, word in ipairs({ 'Lv', 'Level' }) do
		local init = 1
		while true do
			local s, e, lv, colon = text:find(word .. '%s*(%d+)%s*(:?)', init)
			if not s then break end
			init = e + 1
			-- the data sometimes forgets the colon ("Lv20 Name"), accepted at the start of a line
			local lineStart = (s == 1) or text:sub(s - 1, s - 1) == '\n'
			if colon == ':' or lineStart then
				starts[#starts + 1] = { s = s, e = e, lv = lv }
			end
		end
	end
	table.sort(starts, function(a, b) return a.s < b.s end)

	local intro, groups = text, {}
	if #starts > 0 then
		intro = text:sub(1, starts[1].s - 1)
		for i, st in ipairs(starts) do
			local stop = starts[i + 1] and (starts[i + 1].s - 1) or #text
			local spec = trim(text:sub(st.e + 1, stop))
			local names = {}
			for _, n in ipairs(mw.text.split(spec, ' or ', true)) do
				n = trim(n)
				if n ~= '' then names[#names + 1] = n end
			end
			groups[#groups + 1] = { level = st.lv, names = names }
		end
	end

	local lines = {}
	for _, l in ipairs(mw.text.split(intro, '\n', true)) do
		l = trim(l)
		if l ~= '' then lines[#lines + 1] = l end
	end
	return lines, groups
end

-- One skill name with its hover box
-- name: the skill's own name (title and icon of the hover box); shown: the text on the page if it differs
local function skillNode(name, skills, shown)
	local info = skills[name] or {}
	local effect = info.effect
	if effect == '' then effect = nil end

	local s = mw.html.create('span'):addClass('aa-mskill'):attr('tabindex', '0')
	s:wikitext(esc(shown or name))
	local tip = s:tag('span'):addClass('aa-mskill-tip')

	local head = tip:tag('span'):addClass('aa-mskill-head')
	local ext = info.icon or ''
	if EXTERNAL_ICONS and ext:match('^[%w_%-%.]+%.png$') then
		head:tag('span'):addClass('aa-mskill-icon'):wikitext(ICON_BASE .. ext)
	else
		-- the skill's own game icon (icon_skill_....png). No check that the file exists: such checks are
		-- "expensive function calls" (about 100 allowed per page). A skill without icon data gets no icon.
		local fname = IconName.iconFile(ext)
		if fname ~= '' then
			head:tag('span'):addClass('aa-mskill-icon'):wikitext('[[File:' .. fname .. '|40px|link=]]')
		end
	end
	head:tag('span'):addClass('aa-mskill-name'):wikitext(esc(name))

	-- mana, casting time, cooldown (only when the skill data was found)
	local mana = tonumber(info.mana) or 0
	local cd = info.cooldown or ''
	local cast = info.casting or ''
	if cd == 'null' then cd = '' end
	if cast == 'null' or cast == '0' or cast == '0s' then cast = '' end
	if cd ~= '' or cast ~= '' or mana > 0 or info.casting == '0s' then
		local box = tip:tag('span'):addClass('aa-mskill-meta')
		if mana > 0 then
			box:tag('span'):addClass('aa-mskill-metaline'):wikitext('Mana ' .. esc(int(info.mana)))
		end
		if cast == '' then
			box:tag('span'):addClass('aa-mskill-metaline'):wikitext('Instant')
		else
			box:tag('span'):addClass('aa-mskill-metaline'):wikitext('Cast Time ' .. esc(secs(cast)))
		end
		if cd ~= '' then
			box:tag('span'):addClass('aa-mskill-metaline'):wikitext(esc(secs(cd)) .. ' Cooldown')
		end
	end

	if effect then
		tip:tag('span'):addClass('aa-mskill-effect'):wikitext(effectHtml(effect))
	else
		tip:tag('span'):addClass('aa-mskill-none'):wikitext('No description yet.')
	end
	return s
end

-- One hoverable skill name, used by Module:Glider (through Template:GliderSkill).
-- Args: name (the text on the page), realname (the skill's own name from its data: title and icon of the
-- hover box, optional), color (RRGGBB, optional), cd, cast, mana, desc (with colour codes), icon
function p.hover(frame)
	local a = frame.args
	local shown = trim(a.name)
	local name = trim(a.realname)
	if name == '' or name == 'null' then name = shown end
	local skills = { [name] = { effect = trim(a.desc), cooldown = trim(a.cd), casting = trim(a.cast),
		mana = trim(a.mana), icon = trim(a.icon) } }
	local node = skillNode(name, skills, shown)
	local c = trim(a.color):match('^%x%x%x%x%x%x$')
	if c then node:attr('style', 'color:#' .. c .. '; border-bottom-color:#' .. c) end
	return tostring(node)
end

-- A failed API request (with "suppress error" in the template) leaves External Data's own error text
-- behind. Treat that as "no data" instead of showing it.
local function fetched(s)
	s = trim(s)
	if s:find('class="error"', 1, true) then return '' end
	return s
end

-- The skills of a stored mount in the same text form Template:Mount hands over for live data:
-- "level¦name¦effect¦cooldown¦casting¦mana¦effect with colour codes¦icon‖..."
local function storedSkillString(stored)
	local rows = {}
	for _, s in ipairs(stored.skills or {}) do
		local sk = Store.get('Skills', s.id) or {}
		rows[#rows + 1] = table.concat({ s.lv and tostring(s.lv) or '', s.name or '', s.desc or '',
			sk.cd or '', sk.cast or '', sk.mana and tostring(sk.mana) or '', sk.desc or '', sk.icon or '' }, COL) .. ROW
	end
	return table.concat(rows)
end

-- Template:Mount sends the ID (id=) when the data is stored on the wiki. Without it the live data of the
-- API comes in the other arguments.
function p.card(frame)
	local a = frame.args
	local stored = Store.get('Mounts', a.id)
	local name = stored and trim(stored.name) or fetched(a.name)
	if name == '' then
		return '<!-- Mount: no data, the API request failed or the item id is wrong -->'
	end
	local grade = stored and trim(stored.grade) or fetched(a.grade)
	local gradeKnown = grade ~= ''   -- empty = the grade could not be fetched: no grade line
	if grade == '' or grade == 'None' or grade == 'Any' then grade = 'Basic' end   -- mounts without a grade count as Basic
	local speed = fetched(a.speed)   -- a speed written in the template replaces the stored one
	if speed == '' and stored and stored.speed then speed = tostring(stored.speed) end
	local float = trim(a.float):lower()   -- optional: right / left / center
	if float ~= 'right' and float ~= 'left' and float ~= 'center' then float = '' end
	local desc = stored and trim(stored.desc) or fetched(a.desc)
	local skillStr
	if stored then
		skillStr = storedSkillString(stored)
	else
		skillStr = tostring(a.skills or '')   -- read last: expanding it fetches the skill data
		if skillStr:find('class="error"', 1, true) then skillStr = '' end
	end
	local gcls = (grade:lower():gsub('[^%a]', ''))

	-- Colours come from Template:Grade, so there is one place to change them
	local function colored(g, text)
		return frame:expandTemplate{ title = 'Grade', args = { g, text } }
	end

	local icon = IconName.icon({ args = { name, grade, size = 48, file = stored and trim(stored.icon) or fetched(a.icon) } })

	local root = mw.html.create('div'):addClass('aa-mount'):addClass('aa-mgrade-' .. gcls)
	if float ~= '' then root:addClass('aa-mount-' .. float) end

	-- header: icon, category, grade, name
	local head = root:tag('div'):addClass('aa-mount-head')
	head:tag('div'):addClass('aa-mount-icon'):wikitext(icon)
	local title = head:tag('div'):addClass('aa-mount-title')
	title:tag('div'):addClass('aa-mount-cat'):wikitext('[[Mounts|Mount]]')
	if gradeKnown then
		title:tag('div'):addClass('aa-mount-grade'):wikitext(colored(grade, esc(grade)))
	end
	title:tag('div'):addClass('aa-mount-name'):wikitext(colored(grade, esc(name)))

	root:tag('div'):addClass('aa-mount-sep')

	-- description and skills
	local lines, fallbackGroups = parseDescription(desc)
	local groups, skills = parseSkillList(skillStr)
	if #groups == 0 then groups = fallbackGroups end

	-- Some skills have no unlock level in the data (e.g. "Ant's Diligence"), but the description
	-- says which level they belong to. Match them by name (ignoring case, punctuation and doubled
	-- letters, so the description's typo "Dilligence" still matches) and move them into that row.
	local function key(s)
		s = tostring(s):lower():gsub('%f[%a]the%f[%A]', '')   -- "Launch the Owner" = "Launch Owner"
		s = s:gsub('[^%a%d]', '')
		return (s:gsub('(%a)%1', '%1'))   -- doubled letters -> single
	end
	-- Skill names the description lists on a line of their own, before the first level ("Buck Up, Friends!")
	local introKeys = {}
	for _, l in ipairs(lines) do introKeys[key(l)] = true end
	-- sort position of a row: skills listed without a level come first, then by level, level-less ones last
	local function lvl(g)
		if g.keep then return -1 end
		return tonumber(g.level) or 1e9
	end
	-- number of single-letter edits between two strings
	local function distance(a, b)
		local prev = {}
		for j = 0, #b do prev[j] = j end
		for i = 1, #a do
			local cur = { [0] = i }
			for j = 1, #b do
				local cost = (a:sub(i, i) == b:sub(j, j)) and 0 or 1
				cur[j] = math.min(prev[j] + 1, cur[j - 1] + 1, prev[j - 1] + cost)
			end
			prev = cur
		end
		return prev[#b]
	end
	local descLevel, descOrder, descKeys, descGroup, descCount, descFirst = {}, {}, {}, {}, {}, {}
	for gi, dg in ipairs(fallbackGroups) do
		descCount[dg.level] = (descCount[dg.level] or 0) + 1
		if not descFirst[dg.level] then descFirst[dg.level] = gi end
		for i, n in ipairs(dg.names) do
			local k = key(n)
			descLevel[k] = dg.level
			descOrder[k] = i
			descGroup[k] = gi
			descKeys[#descKeys + 1] = k
		end
	end
	-- exact match on the simplified name, otherwise the closest name that is at most 2 letters off
	-- (the description misspells some names, e.g. "Release Poision"). Returns the description's key.
	local function findKey(name)
		local k = key(name)
		if descLevel[k] then return k end
		local best, bestD = nil, 3
		for _, dk in ipairs(descKeys) do
			local d = distance(k, dk)
			if d < bestD then best, bestD = dk, d end
		end
		return best
	end

	-- Several skills of one level are "A or B" (a choice) only when the description writes them on one
	-- line. When it lists the same level on several lines ("Lv20: A" / "Lv20: B", like Carrot Dash),
	-- the mount gets all of them, so every description line becomes its own line here too.
	local split = {}
	for _, g in ipairs(groups) do
		if g.level ~= '' and (descCount[g.level] or 0) > 1 then
			local parts, order = {}, {}
			for _, n in ipairs(g.names) do
				local dk = findKey(n)
				local gi = dk and descGroup[dk] or descFirst[g.level]
				if not parts[gi] then parts[gi] = {}; order[#order + 1] = gi end
				parts[gi][#parts[gi] + 1] = n
			end
			table.sort(order)
			for _, gi in ipairs(order) do
				split[#split + 1] = { level = g.level, names = parts[gi], dg = gi }
			end
		else
			g.dg = descFirst[g.level]
			split[#split + 1] = g
		end
	end
	groups = split

	local merged, byDG = {}, {}
	for _, g in ipairs(groups) do
		if g.level ~= '' then
			if g.dg then byDG[g.dg] = g end
			merged[#merged + 1] = g
		end
	end
	local placed = false
	for _, g in ipairs(groups) do
		if g.level == '' then
			for _, n in ipairs(g.names) do
				local dk = findKey(n)
				local gi = dk and descGroup[dk]
				local target = gi and byDG[gi]
				if target then
					target.names[#target.names + 1] = n
					descOrder[key(n)] = descOrder[dk]
					placed = true
				elseif gi then
					-- the description has a line of its own for this skill (e.g. "Lv20 Powerful Wingbeats")
					local ng = { level = fallbackGroups[gi].level, names = { n }, dg = gi }
					byDG[gi] = ng
					merged[#merged + 1] = ng
					placed = true
				elseif #fallbackGroups > 0 and introKeys[key(n)] then
					-- listed in the description without a level: keep it, as the first row
					merged[#merged + 1] = { level = '', names = { n }, keep = true }
					placed = true
				else
					merged[#merged + 1] = { level = '', names = { n } }
				end
			end
		end
	end
	if placed then
		for _, g in ipairs(merged) do
			table.sort(g.names, function(a, b)
				return (descOrder[key(a)] or 99) < (descOrder[key(b)] or 99)
			end)
		end
		-- keep the rows in level order (same level: description order), level-less ones last
		for i, g in ipairs(merged) do g.idx = i end
		table.sort(merged, function(a, b)
			local la, lb = lvl(a), lvl(b)
			if la ~= lb then return la < lb end
			if (a.dg or 99) ~= (b.dg or 99) then return (a.dg or 99) < (b.dg or 99) end
			return a.idx < b.idx
		end)
		groups = merged
	end

	-- The description lists the mount's skills by level (like the game's tooltip). Then show them in that
	-- order (the API's order is arbitrary) and leave out skills that are on no level and not in the
	-- description (basic skills such as "Mounted Attack", which the game does not list either).
	if #fallbackGroups > 0 then
		local kept = {}
		for _, g in ipairs(groups) do
			if g.level ~= '' or g.keep then kept[#kept + 1] = g end
		end
		if #kept > 0 then
			for i, g in ipairs(kept) do g.idx = i end
			table.sort(kept, function(a, b)
				local la, lb = lvl(a), lvl(b)
				if la ~= lb then return la < lb end
				if (a.dg or 99) ~= (b.dg or 99) then return (a.dg or 99) < (b.dg or 99) end
				return a.idx < b.idx
			end)
			groups = kept
		end
	end

	local isSkill = {}
	for _, g in ipairs(groups) do
		for _, n in ipairs(g.names) do isSkill[key(n)] = true end
	end
	-- The API breaks long descriptions into short lines in the middle of sentences. Glue a line onto
	-- the previous one unless that one ended a sentence (. ! ? : or a closing quote).
	local paras = {}
	for _, l in ipairs(lines) do
		if not isSkill[key(l)] then   -- the description may list the skill names again, we show them below
			local last = paras[#paras]
			if last and not last:find('[%.!%?:"\'%)]$') then
				paras[#paras] = last .. ' ' .. l
			else
				paras[#paras + 1] = l
			end
		end
	end
	for _, para in ipairs(paras) do
		root:tag('div'):addClass('aa-mount-desc'):wikitext(esc(para))
	end
	for _, g in ipairs(groups) do
		local line = root:tag('div'):addClass('aa-mount-skill')
		if g.level ~= '' then
			line:tag('span'):addClass('aa-mount-lv'):wikitext('Lv' .. esc(g.level))
			line:wikitext(': ')
		end
		for i, n in ipairs(g.names) do
			if i > 1 then line:tag('span'):addClass('aa-mount-or'):wikitext(' or ') end
			line:node(skillNode(n, skills))
		end
	end

	-- use, speed
	root:tag('div'):addClass('aa-mount-use'):wikitext('Use:')
	root:tag('div'):addClass('aa-mount-usetext'):wikitext('Summons a mount.')
	if speed ~= '' then
		local line = root:tag('div'):addClass('aa-mount-speed')
		line:wikitext('Base Move Speed ')
		line:tag('span'):addClass('aa-mount-speedval'):wikitext(esc(speed) .. 'm/s')
	end

	return tostring(root)
end

return p
