Module:Tooltip

Source code (Scribunto) as stored on the wiki.

local p = {}

-- Picks the card template for an item ID. Used by Template:Tooltip.
local Store = require('Module:DataStore')

-- NEW CARD TYPE = one line here: the category of the item (category_name in the API) -> the template that shows it.
-- Every category that is not listed uses DEFAULT.
local REGISTRY = {
	Mount = 'Mount',
	Glider = 'Glider',
	Pets = 'Pet',
	['Powerstone Pet'] = 'Pet',
	['Battle Pet'] = 'Pet',
	-- Potion = 'Consumable',   (listed only as an example, DEFAULT already covers it)
}
local DEFAULT = 'Consumable'

local function trim(s)
	return mw.text.trim(tostring(s or ''))
end

-- The category of an item, when the sync script has stored it (nil when not)
local function stored(id)
	local c = Store.get('Cards', id)
	if c and trim(c.category) ~= '' then return trim(c.category) end
	if Store.get('Mounts', id) then return 'Mount' end
	if Store.get('Gliders', id) then return 'Glider' end
	if Store.get('Pets', id) then return 'Pets' end
	return nil
end

-- {{#invoke:Tooltip|stored|ID}} -> "1" when the category is known from the stored data, otherwise nothing
function p.stored(frame)
	return stored(frame.args[1]) and '1' or ''
end

-- {{#invoke:Tooltip|template|id=ID|live=<category from the API>}} -> name of the card template
function p.template(frame)
	local a = frame.args
	local cat = stored(a.id)
	if not cat then
		cat = trim(a.live)
		if cat:find('class="error"', 1, true) then cat = '' end
	end
	return REGISTRY[cat] or DEFAULT
end

return p
