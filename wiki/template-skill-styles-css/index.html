/* Skill card. Position comes inline from Module:Skill. Same look as the Mount / Glider / item cards. */
/* A chain (Triple Slash) has one card per skill inside .aa-skill-set, only one is shown: the requested one
   (.aa-skill-main, always the last in the set), or the one whose icon was clicked. Every card has a small
   anchor in front of it (.aa-skill-anchor, the :target of the icon's link). The anchor is position: fixed, so
   the browser does not scroll the page when it is the target.
   All cards of a set lie on top of each other in one grid cell (the hidden ones are only invisible), so the set
   is as tall as the tallest card and the page does not jump when another skill is shown. The shown card is
   stretched to that height and keeps its Chain row at the bottom. */
.aa-skill-set {
  display: grid;
  justify-items: start;
  max-width: 100%;
}
.aa-skill-anchor {
  position: fixed;
  display: block;
  width: 0;
  height: 0;
  overflow: hidden;
}
.aa-skill-set > .aa-skill {
  grid-area: 1 / 1;
  visibility: hidden;
}
.aa-skill-set > .aa-skill-main {
  visibility: visible;
}
.aa-skill-set > .aa-skill-anchor:target ~ .aa-skill-main {
  visibility: hidden;
}
.aa-skill-set > .aa-skill-anchor:target + .aa-skill {
  visibility: visible;
}
/* the card is a column so that the Chain row can sit at the bottom */
.aa-skill-set > .aa-skill {
  display: flex;
  flex-direction: column;
}
/* empty space where another skill of the chain has "Effect Granted" (same height as the real thing) */
.aa-skill-reserve {
  visibility: hidden;
