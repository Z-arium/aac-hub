Source code (sanitized-css) as stored on the wiki.

/* Ensure the grid item itself stretches */
.tile-row {
  align-items: stretch; /* grid default is stretch, but safe */
}

/* Ensure the halves define a real height for children */
.tile-halves {
  min-height: 300px;
  width: 100%;
  display: grid;         /* better than flex for overlay cards */
}

/* Make the card fill the tile */
#home-featured.home-card {
  position: relative;
  height: 100%;
  min-height: 300px;     /* belt & suspenders */
  overflow: hidden;
  border-radius: var(--border-radius-medium);
}

/* Make background layer fill the card */
#home-featured .home-card__background {
  position: absolute;
  inset: 0;
}

/* Make ANY link inside background fill */
#home-featured .home-card__background a {
  display: block;
  width: 100%;
  height: 100%;
}

/* Make the image fill (covers without distortion) */
#home-featured .home-card__background img {
  display: block;
  width: 100%;
  height: 100%;
  object-fit: cover;
}

/* Gradient overlay stays on top of background */
#home-featured .home-card__background::after {
  content: "";
  position: absolute;
  left: 0;
  right: 0;
  bottom: 0;
  height: 60%;
  background: linear-gradient(to top, #000, transparent);
  transition: transform 0.2s ease;
}

/* Foreground overlay fills too */
#home-featured .home-card__foreground {
  position: absolute;
  inset: 0;
  display: flex;
  flex-direction: column;
  justify-content: flex-end;
  pointer-events: none; /* optional: foreground not clickable */
}
