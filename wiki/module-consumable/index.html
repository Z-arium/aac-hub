Module:Consumable

Source code (Scribunto) as stored on the wiki.

local p = {}

-- In-game style card for items that are used up or opened: potions, food, talismans, tokens, crates and
-- every other item that has no card of its own. The category label (Potion, Food ...) comes from the API.
local IconName = require('Module:IconName')
local Store = require('Module:DataStore')   -- data stored on the wiki by sync_wiki_data.py

local function trim(s)
	return mw.text.trim(tostring(s or ''))
end

local function esc(s)
	return mw.text.encode(tostring(s or ''), '<>&"\'%[%]{}|')
end

-- Colours of the game's short codes: |nc;...|r is orange, |ni;...|r is yellow, |nd;...|r is light blue ("loot drop rate"). Others show without colour.
local NCODE = { c = 'FF9C27', i = 'F5CB65', d = 'A3E3FD' }

-- The game marks highlighted words as |cFFRRGGBB...|r; only those parts get a colour.
-- A coloured part may run over several lines: every line gets its own span, so lines can be split afterwards.
local function coloured(text, hook)
	text = tostring(text or '')
	-- hook(code, inner) may return ready HTML for a |nX;...|r part (the pet card puts hoverable skills there)
	local toks = {}
	text = text:gsub('|n(%a);(.-)|r', function(code, inner)
		if hook then
			local h = hook(code, inner)
			if h then
				toks[#toks + 1] = h
				return '\1' .. #toks .. '\2'
			end
		end
		if NCODE[code] then return '|cFF' .. NCODE[code] .. inner .. '|r' end
		return inner
	end)
	local out, pos = {}, 1
	while true do
		local s, e, rgb, inner = text:find('|c%x%x(%x%x%x%x%x%x)(.-)|r', pos)
		if not s then break end
		out[#out + 1] = esc(text:sub(pos, s - 1))
		local parts = {}
		for _, seg in ipairs(mw.text.split(inner, '\n', true)) do
			if mw.text.trim(seg) == '' then
				parts[#parts + 1] = seg
			else
				parts[#parts + 1] = '<span style="color:#' .. rgb .. '">' .. esc(seg) .. '</span>'
			end
		end
		out[#out + 1] = table.concat(parts, '\n')
		pos = e + 1
	end
	out[#out + 1] = esc(text:sub(pos))
	local html = table.concat(out)
	if #toks > 0 then
		html = html:gsub('\1(%d+)\2', function(i) return toks[tonumber(i)] end)
	end
	return html
end

-- Where the card sits on the page (float parameter), set inline so it works without the style page
local FLOATS = {
	right = 'float:right; clear:right; margin:0 0 0.5em 1em;',
	left = 'float:left; clear:left; margin:0 1em 0.5em 0;',
	center = 'margin:0 auto 0.5em auto;',
}

-- Binding of the item (bind_type of the API). "normal" shows nothing.
local BIND = {
	soulbound_pickup = 'Untradeable',
	soulbound_unpack = 'Untradeable',
	soulbound_equip = 'Binds when equipped',
}

-- "-@ITEM_NAME(38620)" in a description is the name of that item
local function names(text)
	return (text:gsub('@ITEM_NAME%((%d+)%)', function(id)
		local it = Store.get('Items', id) or Store.get('Cards', id)
		return it and trim(it.name) ~= '' and trim(it.name) or ('Item ' .. id)
	end))
end

-- Text with line breaks: every line its own row, a blank line leaves some space
local function paragraphs(text, class, hook)
	text = trim(names(tostring(text or ''):gsub('\r', '')))
	if text == '' then return nil end
	local box = mw.html.create('div'):addClass(class)
	local gap = false
	for _, line in ipairs(mw.text.split(coloured(text, hook), '\n', true)) do
		line = trim(line)
		if line == '' then
			gap = true
		else
			local row = box:tag('div'):wikitext(line)
			if gap then row:addClass('aa-item-gap') end
			gap = false
		end
	end
	return box
end

local function fetched(s)
	s = trim(s)
	if s:find('class="error"', 1, true) then return '' end
	return s
end

-- shared with Module:Skill
p.paragraphs = paragraphs

-- The card from ready data d = { name, grade, category, icon, over, desc, use, bind, req, nosell }
-- opts: float, hook (see coloured), use (replaces the Use text), bind (true = show "Untradeable"), nosell (false = hide "Cannot Sell")
function p.render(frame, d, opts)
	opts = opts or {}
	local name = trim(d.name)
	local grade = trim(d.grade)
	local hasGrade = not (grade == '' or grade == 'None' or grade == 'Any')
	if not hasGrade then grade = 'Basic' end   -- no grade: Basic look, and no grade line
	local category = trim(d.category)
	local use = trim(opts.use)
	if use == '' then use = trim(d.use) end
	local bind = BIND[trim(d.bind)]

	local float = trim(opts.float):lower()
	if not FLOATS[float] then float = '' end
	local root = mw.html.create('div'):addClass('aa-item')
	if float == 'right' or float == 'left' then root:addClass('aa-item-float') end
	if float == 'right' then root:addClass('aa-item-right') end
	if float ~= '' then root:attr('style', FLOATS[float]) end

	-- header: icon, category, grade, name
	local head = root:tag('div'):addClass('aa-item-head')
	head:tag('div'):addClass('aa-item-icon'):wikitext(IconName.icon({ args = { name, grade, size = 48, file = d.icon, over = d.over } }))
	local title = head:tag('div'):addClass('aa-item-title')
	if category ~= '' then title:tag('div'):addClass('aa-item-cat'):wikitext(esc(category)) end
	if hasGrade then
		title:tag('div'):addClass('aa-item-grade'):wikitext(
			frame:expandTemplate{ title = 'Grade', args = { grade, esc(grade) } })
	end
	title:tag('div'):addClass('aa-item-name'):wikitext(
		frame:expandTemplate{ title = 'Grade', args = { grade, esc(name) } })

	root:tag('div'):addClass('aa-item-sep')

	-- "Untradeable" only where the card asks for it (opts.bind); the required level is not shown
	if bind and opts.bind then root:tag('div'):addClass('aa-item-line'):wikitext(esc(bind)) end
	local body = paragraphs(d.desc, 'aa-item-desc', opts.hook)
	if body then root:node(body) end

	local useBox = paragraphs(use, 'aa-item-green')
	if useBox then
		root:tag('div'):addClass('aa-item-label'):wikitext('Use:')
		root:node(useBox)
	end
	if d.nosell and opts.nosell ~= false then root:tag('div'):addClass('aa-item-nosell'):wikitext('Cannot Sell') end
	return tostring(root)
end

function p.card(frame)
	local a = frame.args
	local d = Store.get('Cards', a.id)
	local name = d and trim(d.name) or fetched(a.name)
	if name == '' then
		return '<!-- Consumable: no data, the API request failed or the item id is wrong -->'
	end
	local data
	if d then
		data = d
	else
		-- live data: sellable comes as true/false (or 1/0), the refund is the price the shop pays
		local sell = fetched(a.csell):lower()
		data = { name = name, grade = fetched(a.grade), category = fetched(a.category), icon = fetched(a.icon),
			over = fetched(a.over), desc = fetched(a.desc), use = fetched(a.cuse), bind = fetched(a.bind),
			req = fetched(a.req), nosell = (sell == 'false' or sell == '0' or fetched(a.crefund) == '0') }
	end
	return p.render(frame, data, { float = a.float, use = a.use, bind = trim(a.untradeable):lower() ~= 'no',
		nosell = trim(a.cannotsell):lower() ~= 'no' })
end

return p
